async function executeChunkedInsert(table, columns, data, conflictClause) {
    if (data.length === 0) return { inserted: 0, updated: 0, unchanged: 0 };
    let inserted = 0, updated = 0, unchanged = 0;
    for (let i = 0; i < data.length; i += 2000) {
        const chunk = data.slice(i, i + 2000);
        const values = chunk.map(r => `(${r.map(v => v === null ? 'NULL' : `'${String(v).replace(/'/g, "''")}'`).join(',')})`);
        const query = `
            WITH rows AS (
                INSERT INTO ${table} (${columns.join(', ')}) VALUES ${values.join(',')} 
                ${conflictClause} RETURNING (xmax = 0) AS is_insert
            )
            SELECT count(*) FILTER (WHERE is_insert) AS ins, count(*) FILTER (WHERE NOT is_insert) AS upd FROM rows;
        `;
        try {
            const res = await client.query(query);
            const ins = parseInt(res.rows[0].ins || 0);
            const upd = parseInt(res.rows[0].upd || 0);
            inserted += ins; updated += upd; unchanged += (chunk.length - ins - upd);
        } catch(e) {
            unchanged += chunk.length;
        }
    }
    return { inserted, updated, unchanged };
}
